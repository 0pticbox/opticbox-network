  connectionBadge.textContent="LIVE";
  connectionBadge.classList.add("connected");
  setStatus("connected",`${mode.toUpperCase()} CONNECTED`,`Source: ${name}`);
}

async function connectDesktopAudio() {
  try {
    stopAudio(false);
    if (!navigator.mediaDevices?.getDisplayMedia) {
      throw new Error("Desktop audio capture is not supported in this browser.");
    }

    setStatus("warning","SELECT AN AUDIO SOURCE","Choose a music tab and enable Share tab audio.");

    captureStream=await navigator.mediaDevices.getDisplayMedia({
      video:true,
      audio:true,
      preferCurrentTab:false,
      selfBrowserSurface:"exclude",
      surfaceSwitching:"include",
      systemAudio:"include"
    });

    const audioTracks=captureStream.getAudioTracks();
    if (!audioTracks.length) {
      throw new Error("No audio track was received. Use microphone or audio-file mode in Firefox.");
    }

    await prepareAudioAnalyser();
    const audioTrack=audioTracks[0];
    const audioOnly=new MediaStream([audioTrack]);
    sourceNode=audioContext.createMediaStreamSource(audioOnly);
    sourceNode.connect(analyser);

    captureStream.getTracks().forEach(track=>{
      track.addEventListener("ended",()=>stopAudio());
    });

    finishConnection("desktop audio",audioTrack.label||"shared browser audio");
  } catch (error) {
    console.error(error);
    stopAudio(false);
    connectionBadge.textContent="ERROR";
    setStatus("warning","DESKTOP AUDIO FAILED",`${error.name||"Error"}: ${error.message}`);
  }
}

async function connectMicrophone() {
  try {
    stopAudio(false);
    if (!navigator.mediaDevices?.getUserMedia) {
      throw new Error("Microphone input is not supported.");
    }

    setStatus("warning","MICROPHONE PERMISSION","Allow microphone access.");

    captureStream=await navigator.mediaDevices.getUserMedia({
      audio:{
        echoCancellation:false,
        noiseSuppression:false,
        autoGainControl:false
      },
      video:false
    });

    await prepareAudioAnalyser();
    sourceNode=audioContext.createMediaStreamSource(captureStream);
    sourceNode.connect(analyser);

    const track=captureStream.getAudioTracks()[0];
    if (!track) throw new Error("No microphone track was received.");
    track.addEventListener("ended",()=>stopAudio());

    finishConnection("microphone",track.label||"microphone");
  } catch (error) {
    console.error(error);
    stopAudio(false);
    connectionBadge.textContent="ERROR";
    setStatus("warning","MICROPHONE FAILED",`${error.name||"Error"}: ${error.message}`);
  }
}

async function loadAudioFile(file) {
  try {
    if (!file) return;
    if (!file.type.startsWith("audio/")&&!/\.(mp3|wav|ogg|m4a|aac|flac)$/i.test(file.name)) {
      throw new Error("Choose a supported audio file.");
    }

    stopAudio(false);
    if (currentObjectUrl) URL.revokeObjectURL(currentObjectUrl);
    currentObjectUrl=URL.createObjectURL(file);

    audioPlayer.src=currentObjectUrl;
    audioPlayer.classList.add("has-file");
    fileName.textContent=file.name;

    await prepareAudioAnalyser();

    if (!mediaElementSource) {
      mediaElementSource=audioContext.createMediaElementSource(audioPlayer);
    }

    try { mediaElementSource.disconnect(); } catch (_) {}
    mediaElementSource.connect(analyser);
    analyser.connect(audioContext.destination);

    finishConnection("audio file",file.name);
    await audioPlayer.play();
  } catch (error) {
    console.error(error);
    stopAudio(false);
    connectionBadge.textContent="ERROR";
    setStatus("warning","FILE FAILED",`${error.name||"Error"}: ${error.message}`);
  }
}

function stopAudio(showMessage=true) {
  connected=false;

  if (captureStream) {
    captureStream.getTracks().forEach(track=>track.stop());
    captureStream=null;
  }

  if (sourceNode) {
    try { sourceNode.disconnect(); } catch (_) {}
    sourceNode=null;
  }

  if (mediaElementSource) {
    try { mediaElementSource.disconnect(); } catch (_) {}
  }

  if (analyser) {
    try { analyser.disconnect(); } catch (_) {}
  }

  audioPlayer.pause();
  analyser=null;
  frequencyData=null;
  timeData=null;
  detectedLevel=0;
  bassLevel=0;
  trebleLevel=0;

  levelMeter.style.width="0%";
  levelNumber.textContent="0%";

  connectButton.disabled=false;
  micButton.disabled=false;
  fileButton.disabled=false;
  stopButton.disabled=true;
  connectionBadge.textContent="OFFLINE";
  connectionBadge.classList.remove("connected");

  if (showMessage) {
    setStatus("offline","AUDIO DISCONNECTED","Choose an audio source and visual mode.");
  }
}
