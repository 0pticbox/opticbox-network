function setScene(scene) {
  visualModeSelect.value = scene;
  lastSceneChange = performance.now();

function cycleScene(direction=1) {
  const index = SCENES.indexOf(visualModeSelect.value);
  setScene(SCENES[(index + direction + SCENES.length) % SCENES.length]);
}

function randomizeScene() {
  let next = visualModeSelect.value;
  while (next === visualModeSelect.value && SCENES.length > 1) {
    next = SCENES[Math.floor(Math.random() * SCENES.length)];
  }
  setScene(next);
  sceneSpeedSlider.value = (0.6 + Math.random() * 1.8).toFixed(1);
  trailAmountSlider.value = (0.08 + Math.random() * 0.28).toFixed(2);
  updateLabels();
}

function maybeAutoCycle(now) {
  if (!autoCycleCheckbox.checked) return;
  const wait = Number(cycleSecondsSlider.value) * 1000;
  if (now - lastSceneChange >= wait) cycleScene(1);
}

function analyzeAudio() {
  if (!analyser || !frequencyData || !timeData) {
    detectedLevel = 0;
    bassLevel = 0;
    trebleLevel = 0;
    levelMeter.style.width = "0%";
    levelNumber.textContent = "0%";
    return;
  }

  analyser.getByteFrequencyData(frequencyData);
  analyser.getByteTimeDomainData(timeData);

  let rmsSum = 0;
  for (let i=0;i<timeData.length;i++) {
    const n = (timeData[i]-128)/128;
    rmsSum += n*n;
  }

  detectedLevel = Math.min(
    1,
    Math.sqrt(rmsSum/timeData.length) * Number(sensitivitySlider.value) * 4.5
  );

  const bassEnd = Math.max(3, Math.floor(frequencyData.length*0.04));
  const trebleStart = Math.floor(frequencyData.length*0.28);
  const trebleEnd = Math.floor(frequencyData.length*0.68);

  let bassSum = 0;
  for (let i=0;i<bassEnd;i++) bassSum += frequencyData[i];
  bassLevel = Math.min(1, (bassSum/bassEnd/255) * Number(bassBoostSlider.value));

  let trebleSum = 0;
  let trebleCount = 0;
  for (let i=trebleStart;i<trebleEnd;i++) {
    trebleSum += frequencyData[i];
    trebleCount++;
  }
  trebleLevel = trebleCount ? trebleSum/trebleCount/255 : 0;

  const percent = Math.round(detectedLevel*100);
  levelMeter.style.width = `${percent}%`;
  levelNumber.textContent = `${percent}%`;
}

function fadeBackground(alpha=0.18,color=COLORS.black) {
  ctx.fillStyle = hexWithAlpha(color,alpha);
  ctx.fillRect(0,0,WIDTH,HEIGHT);
}

function hexWithAlpha(hex,alpha) {
  const a = Math.round(Math.max(0,Math.min(1,alpha))*255).toString(16).padStart(2,"0");
  return `${hex}${a}`;
}

function getFreqValue(ratio) {
  if (!frequencyData) return 0;
  const index = Math.min(
    frequencyData.length-1,
    Math.floor(Math.pow(ratio,1.8)*frequencyData.length*0.72)
  );
  return frequencyData[index]/255;
}

function drawSpectrum() {
  ctx.fillStyle = COLORS.black;
  ctx.fillRect(0,0,WIDTH,HEIGHT);

  for (let i=0;i<40;i++) {
    const x = (i*83 + visualTime*0.35) % WIDTH;
    const y = (i*47) % 330;
    ctx.fillStyle = i%3 ? COLORS.darkBlue : COLORS.purple;
    ctx.fillRect(x,y,i%7===0?4:2,i%7===0?4:2);
  }

  ctx.strokeStyle = COLORS.lavender;
  ctx.lineWidth = 2;

  for (let y=400;y<HEIGHT;y+=22) {
    ctx.beginPath();
    ctx.moveTo(0,y);
    ctx.lineTo(WIDTH,y);
    ctx.stroke();
  }

  for (let x=-WIDTH;x<WIDTH*2;x+=42) {
    ctx.beginPath();
    ctx.moveTo(WIDTH/2,390);
    ctx.lineTo(x+(visualTime*1.5)%42,HEIGHT);
    ctx.stroke();
  }

  const count = Number(barCountSlider.value);
  while (displayedBars.length<count) {
    displayedBars.push(0);
    peakBars.push(0);
  }
  displayedBars.length = count;
  peakBars.length = count;

  const margin = 24;
  const gap = count>=48?2:4;
  const width = (WIDTH-margin*2-gap*(count-1))/count;
  const floorY = 380;

  for (let i=0;i<count;i++) {
    const target = getFreqValue(i/count)*285*Number(sensitivitySlider.value);
    if (target>displayedBars[i]) displayedBars[i] += (target-displayedBars[i])*.7;
    else displayedBars[i] *= Number(fallSpeedSlider.value);

    const h = Math.max(3,displayedBars[i]);
    peakBars[i] = h>peakBars[i]?h:Math.max(0,peakBars[i]-1.5);

    const x = margin+i*(width+gap);
    let color = COLORS.blue;
    if (h>230) color=COLORS.red;
    else if (h>170) color=COLORS.orange;
    else if (h>105) color=COLORS.yellow;
    else if (h>50) color=COLORS.green;

    ctx.fillStyle = color;
    ctx.fillRect(x,floorY-h,Math.max(1,width),h);
    ctx.fillStyle = COLORS.white;
    ctx.fillRect(x,floorY-h,Math.max(1,width),3);
    ctx.fillStyle = COLORS.pink;
    ctx.fillRect(x,floorY-peakBars[i],Math.max(1,width),4);
